// Improved function to parse shareholding patterns response
function parseShareholdingPatterns(response) {
    if (!response || typeof response !== "string") {
        console.error("Invalid response format for parsing");
        return [];
    }

    const lines = response.split("\n").map(line => line.trim()).filter(line => line.length > 0);
    const companies = [];
    let currentCompany = null;

    lines.forEach(line => {
        const companyMatch = line.match(/^(.+):$/); // Detect "Company Name:"
        if (companyMatch) {
            if (currentCompany) companies.push(currentCompany);
            currentCompany = { name: companyMatch[1], promoterHolding: 0, retailHolding: 0, fiiHolding: 0, othersHolding: 0 };
        } else if (currentCompany) {
            if (/Promoter/i.test(line)) currentCompany.promoterHolding = extractPercentage(line);
            else if (/Retail/i.test(line)) currentCompany.retailHolding = extractPercentage(line);
            else if (/FII/i.test(line)) currentCompany.fiiHolding = extractPercentage(line);
            else if (/Others/i.test(line)) currentCompany.othersHolding = extractPercentage(line);
        }
    });

    if (currentCompany) companies.push(currentCompany);
    return companies;
}

        const finalData = {

            topCompanies,
