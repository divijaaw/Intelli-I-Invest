// Function to make API requests
async function fetchGroqResponse(prompt) {
    try {
        const response = await axios.post(
            API_URL,
            {
                model: "llama3-8b-8192",
                messages: [{ role: "user", content: prompt }],
                max_tokens: 500,
            },
            {
                headers: {
                    Authorization: `Bearer ${API_KEY}`,
                    "Content-Type": "application/json",
                },
            }
        );

        const content = response.data.choices[0].message.content.trim();
        console.log(`Groq API Response for "${prompt}":`, content);
        return content;
    } catch (error) {
        console.error("Error calling Groq API:", error.response?.data || error.message);
        return null;
    }
}

// Function to extract percentage values from text
function extractPercentage(text) {
    const match = text.match(/(\d+(\.\d+)?)%/);
    return match ? parseFloat(match[1]) : 0;
}

// Function to parse shareholding patterns response
function parseShareholdingPatterns(response) {
    if (!response || typeof response !== "string") {
        console.error("Invalid response format for parsing");
        return [];
    }

    const lines = response.split("\n").map(line => line.trim()).filter(line => line.length > 0);
    const companies = [];
    let currentCompany = null;

    lines.forEach((line) => {
        const companyMatch = line.match(/^(.+): Promoter (\d+(\.\d+)?)%/);
        if (companyMatch) {
            if (currentCompany) companies.push(currentCompany);
            currentCompany = {
                name: companyMatch[1],
                promoterHolding: parseFloat(companyMatch[2]),
                retailHolding: 0,
                fiiHolding: 0,
                othersHolding: 0,
                priorityScore: 0
            };
        } else if (currentCompany) {
            if (/Retail/i.test(line)) currentCompany.retailHolding = extractPercentage(line);
            else if (/FII/i.test(line)) currentCompany.fiiHolding = extractPercentage(line);
            else if (/Others/i.test(line)) currentCompany.othersHolding = extractPercentage(line);
        }
    });

    if (currentCompany) companies.push(currentCompany);
    return companies;
}

// API Endpoint to fetch and process tyre sector data
app.get("/tyreSector", async (req, res) => {
    const sector = "tyre";

    try {
        console.log(`Fetching data for sector: ${sector}`);

        // Step 1: Get Market Shares of Companies
        const marketShares = await fetchGroqResponse(
            `Provide the market share percentages of companies in the ${sector} sector in India, formatted as "Company Name: X%" per line.`
        );
        if (!marketShares) throw new Error("Failed to fetch market shares");

        // Step 2: Get Top Companies
        const topCompanies = await fetchGroqResponse(
            `List only the names of companies in the ${sector} sector in India that have a market share greater than 5%.`
        );
        if (!topCompanies) throw new Error("Failed to fetch top companies");

        // Step 3: Get Shareholding Patterns
        const shareholdingPatternsResponse = await fetchGroqResponse(
            `List the shareholding patterns of the top companies in the ${sector} sector in India. 
            Format each company as: "Company Name: Promoter X%, Retail Y%, FII Z%, Others W%" per line.`
        );
        if (!shareholdingPatternsResponse) throw new Error("Failed to fetch shareholding patterns");

        // Step 4: Parse Shareholding Patterns
        let companies = parseShareholdingPatterns(shareholdingPatternsResponse);
        if (!companies.length) throw new Error("No valid company data extracted");

        // Step 5: Calculate Priority Scores
        companies = companies.map((company) => {
            let priorityScore = 0;
            if (company.promoterHolding > 50) priorityScore += 2;
            if (company.retailHolding > 30) priorityScore += 1;
            if (company.fiiHolding > 10) priorityScore += 1;
            if (company.othersHolding > 20) priorityScore -= 1;
            return { ...company, priorityScore };
        });

        // Step 6: Sort companies by priority score (Descending)
        companies.sort((a, b) => b.priorityScore - a.priorityScore);

        // Step 7: Filter and Sort Top Priority Companies (Ascending)
        const topPriorityCompanies = companies
            .filter(company => company.priorityScore > 0)
            .sort((a, b) => a.priorityScore - b.priorityScore);

        // Step 8: Final JSON Response
        const finalData = {
            sector,
            marketShares,
            topCompanies,
            shareholdingPatterns: companies,
            topPriorityCompanies
        };

        // Save data to JSON file
        fs.writeFileSync("final_output.json", JSON.stringify(finalData, null, 2));

        // Send response
        res.json(finalData);
    } catch (error) {
        console.error("Error:", error.message);
        res.status(500).json({ error: error.message });
    }
});
