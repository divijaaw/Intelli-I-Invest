// Function to fetch companies from API with the "upto" parameter
async function getCompanies(upto = 5) {
    try {
        const response = await axios.post('https://nifty-500-stock-market-data-api-nse-india.p.rapidapi.com/', {
            jsonBody: {
                function_name: 'nifty_stock_api',
                upto: upto.toString() // Dynamically set "upto" based on the request
            }
        }, {
            headers: {
                'x-rapidapi-key': RAPIDAPI_KEY,
                'x-rapidapi-host': 'nifty-500-stock-market-data-api-nse-india.p.rapidapi.com',
                'Content-Type': 'application/json'
            }
        });

        return response.data.message.stocks_data; // Get stocks data from the response
    } catch (error) {
        console.error("Error fetching companies:", error);
        return [];
    }
}

// Function to filter companies based on market share (ROE > 10%)
async function filterCompanies(upto) {
    try {
        const companies = await getCompanies(upto);

        // Filter companies with ROE greater than 10% (adjusting for your requirement)
        const filteredCompanies = companies.filter(company => parseFloat(company.roe) > 10);

        return filteredCompanies;
    } catch (error) {
        console.error("Error filtering companies:", error);
        return [];
    }
}

// Existing sectorParticipants route - Now with dynamic sector input and company fetching
app.get('/sectorParticipants', async (req, res) => {
    const { sector, upto } = req.query; // Get sector and upto from query parameters

    if (!sector) {
        return res.status(400).json({ error: 'Sector is required!' });
    }

    try {
        // Fetch and filter companies based on dynamic sector input
        const companies = await filterCompanies(upto || 5); // Default to top 5 if "upto" is not provided
        
        // If no companies are found, respond with an appropriate message
        if (companies.length === 0) {
            return res.status(404).json({ message: 'No companies found with the specified criteria.' });
        }

        // Respond with the filtered companies
        res.json({ companies });

    } catch (error) {
        console.error('Error:', error.message);
        res.status(500).json({ error: 'An error occurred while processing the request.' });
    }
});
