// Function to fetch companies based on dynamic sector input
async function getCompaniesInSector(sector) {
    try {
        const response = await axios.get('https://nifty-500-api.p.rapidapi.com/stocks', {
            params: { sector },
            headers: { 'X-RapidAPI-Key': RAPIDAPI_KEY }
        });

        return response.data.filter(company => company.marketShare > 5); // Fetch companies with market share greater than 5%
    } catch (error) {
        console.error("Error fetching companies:", error);
        return [];
    }
}

// Existing sectorParticipants route - Now with dynamic sector input and company fetching
app.get('/sectorParticipants', async (req, res) => {
    const { sector } = req.query; // Get sector from query parameters

    if (!sector) {
        return res.status(400).json({ error: 'Sector is required!' });
    }

    try {
        // Fetch companies based on dynamic sector input
        const companies = await getCompaniesInSector(sector);
        
        // If no companies are found, respond with an appropriate message
        if (companies.length === 0) {
            return res.status(404).json({ message: 'No companies found with a market share greater than 5%' });
        }

        // Respond with the filtered companies
        res.json({ companies });

    } catch (error) {
        console.error('Error:', error.message);
        res.status(500).json({ error: 'An error occurred while processing the request.' });
    }
});
