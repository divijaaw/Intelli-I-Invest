// Function to filter companies based on given rules
function filterCompanies(companies) {
    return companies
        .map(company => {
            let priorityScore = 0;

            // Apply filtering logic
            if (company.promoterHolding > 50) priorityScore += 2;
            if (company.retailHolding > 30) priorityScore += 1;
            if (company.fiiHolding > 10) priorityScore -= 1; // FII heavy -> negative impact
            if (company.othersHolding > 20) priorityScore -= 1; // Others high -> suspicious

            return { ...company, priorityScore };
        })
        .sort((a, b) => b.priorityScore - a.priorityScore); // Sort by priority
}

// Main function to process all PDFs
async function processShareholdingData() {
    const companyData = [];

    // Read all PDF files from the folder
    const pdfFiles = fs.readdirSync(PDF_FOLDER).filter(file => file.endsWith(".pdf"));

    for (const pdfFile of pdfFiles) {
        const pdfPath = path.join(PDF_FOLDER, pdfFile);
        console.log(`Processing ${pdfFile}...`);

        const text = await extractTextFromPDF(pdfPath);
        const shareholdingData = parseShareholdingData(text);

        companyData.push({
            name: pdfFile.replace(".pdf", ""),
            ...shareholdingData,
        });
    }

    const filteredCompanies = filterCompanies(companyData);

    console.log("Filtered and Ranked Companies:", filteredCompanies);
    return filteredCompanies;
}

// Run the process
processShareholdingData();

const fs = require("fs");
const pdf = require("pdf-parse");

async function extractShareholdingData(pdfPath) {
    const dataBuffer = fs.readFileSync(pdfPath);
    const data = await pdf(dataBuffer);
    const text = data.text; // Extracted raw text

    // Regex patterns to capture relevant shareholding data
    const promoterMatch = text.match(/\(A\) Promoter & Promoter Group.*?(\d+\.\d+)/);
    const publicMatch = text.match(/\(B\) Public.*?(\d+\.\d+)/);
    const nonPromoterMatch = text.match(/\(C\) Non Promoter-Non Public.*?(\d+\.\d+)/);

    return {
        promoterHolding: promoterMatch ? parseFloat(promoterMatch[1]) : null,
        publicHolding: publicMatch ? parseFloat(publicMatch[1]) : null,
        nonPromoterHolding: nonPromoterMatch ? parseFloat(nonPromoterMatch[1]) : null
    };
}
